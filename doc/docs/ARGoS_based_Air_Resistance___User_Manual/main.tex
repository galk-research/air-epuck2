\section{Introduction}
This project extends ARGoS with a simple but effective wind model and aerodynamic ``blocking'' (wake) between robots.
It demonstrates how an upwind robot can shield a downwind robot, allowing the downwind robot to make upwind progress even when an unshielded robot cannot.

\subsection{What you get}
\begin{itemize}
  \item \textbf{Global wind}: a 2D wind vector set once per experiment (\texttt{angle\_deg} and \texttt{magnitude} in \textbf{cm/s}).
  \item \textbf{Post-step impulses}: wind and drive are applied after collision resolution, so the physics engine stays authoritative.
  \item \textbf{Wake / blocking}: upwind robots reduce effective wind for downwind neighbors, with smooth lateral falloff and downwind fade.
  \item \textbf{Qt overlay}: a wind arrow visualization in Qt-OpenGL.
  \item \textbf{Consistent units}: controller \texttt{velocity} and wind \texttt{magnitude} are both in \textbf{cm/s}.
\end{itemize}

\section{Repository Layout \& Quick Start}

\subsection{Repository layout (high level)}
\begin{itemize}
  \item \texttt{src/controllers/air\_resistance/} --- \textbf{main contributed code}: base controller \texttt{CAirResistance} (wind + wake pipeline)
  \item \texttt{src/loop\_functions/wind\_loop\_functions/} --- \textbf{loop functions + Qt user functions}: wind arrow overlay
  \item \texttt{examples/experiments/} --- ready-to-run \texttt{.argos} experiments
  \item \texttt{examples/controllers/} --- example controllers that \textbf{inherit from} \texttt{CAirResistance}
  \item \texttt{doc/} --- Doxygen configuration + LaTeX sources (manuals)
\end{itemize}

\subsection{Prerequisites (install before building/running)}
Before building this repository, you must have:

\begin{itemize}
  \item \textbf{ARGoS 3} installed \textbf{with the \texttt{dynamics2d} physics engine available} (Chipmunk backend).
        ARGoS \textbf{3.0.0-beta or newer} is recommended.
  \item Standard build tools: \textbf{CMake}, \textbf{make} (or Ninja), and a \textbf{C++ compiler}.
  \item \textbf{Robot plugins} depending on which experiments you want to run:
    \begin{itemize}
      \item \textbf{foot-bot}: typically included with standard ARGoS distributions.
      \item \textbf{e-puck2}: must be installed separately (upstream plugin). This repo does \emph{not} ship the upstream e-puck2 robot plugin.
            If you want to run the e-puck2 experiments under \texttt{examples/experiments/}, install the e-puck2 plugin and make sure ARGoS can load it.
    \end{itemize}
\end{itemize}

\subsection{Quick start (recommended path)}
From the repository root:
\begin{lstlisting}
mkdir -p build
cd build
cmake ..
make -j
\end{lstlisting}

Run an example (also from the repository root):
\begin{lstlisting}
argos3 -c examples/experiments/airResistance_two_with_block.argos
\end{lstlisting}

\paragraph{Notes}
\begin{itemize}
  \item Example \texttt{.argos} files load controller/loop-function libraries from \texttt{build/lib/...} using relative paths.
  \item If you want to skip building example controllers:
\begin{lstlisting}
cmake -DBUILD_EXAMPLE_CONTROLLERS=OFF ..
\end{lstlisting}
\end{itemize}

\section{Configuration}

\subsection{Global wind}
Wind is configured once per experiment:
\begin{lstlisting}
<configuration>
  <air_resistance angle_deg="0" magnitude="15.0"/>
</configuration>
\end{lstlisting}

\subsection{Wake (blocking) tuning via XML}
The wake model is tuned from the same \texttt{<air\_resistance .../>} tag.

\subsubsection*{Units}
\begin{itemize}
  \item \texttt{lateral\_reach\_radii}, \texttt{shadow\_length\_radii}, \texttt{upwind\_gate\_radii}: \textbf{dimensionless} (multiples of the blocking robot radius)
  \item \texttt{gamma\_boost}: \textbf{dimensionless}
  \item \texttt{adv\_radius\_min\_m}, \texttt{adv\_radius\_max\_m}: \textbf{meters}
\end{itemize}

\subsubsection*{Defaults (if omitted)}
\begin{table}[H]
\centering
\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}ll@{}}
\toprule
\textbf{Attribute} & \textbf{Default}\\
\midrule
\texttt{lateral\_reach\_radii} & 3.0\\
\texttt{shadow\_length\_radii} & 4.0\\
\texttt{gamma\_boost}          & 2.0\\
\texttt{upwind\_gate\_radii}   & 0.5\\
\texttt{adv\_radius\_min\_m}   & 0.005\\
\texttt{adv\_radius\_max\_m}   & 0.20\\
\bottomrule
\end{tabular}
\end{table}

\subsubsection*{Example (copy/paste)}
\begin{lstlisting}
<configuration>
  <air_resistance
    angle_deg="0"
    magnitude="15.0"
    lateral_reach_radii="3.0"
    shadow_length_radii="4.0"
    gamma_boost="2.0"
    upwind_gate_radii="0.5"
  />
</configuration>
\end{lstlisting}

\subsubsection*{How the knobs affect behavior}
\begin{itemize}
  \item \texttt{shadow\_length\_radii} (\textbf{length}): larger $\Rightarrow$ wake extends further downwind.
  \item \texttt{lateral\_reach\_radii} (\textbf{width}): larger $\Rightarrow$ wake covers larger sideways offsets.
  \item \texttt{gamma\_boost} (\textbf{shape}): 1.0 disables boost; larger values emphasize mid-range shielding (stronger wake core).
  \item \texttt{upwind\_gate\_radii} (\textbf{behindness gate}): prevents side-by-side neighbors from being counted as blockers.
\end{itemize}

\subsection{Controller block and RAB notes}
\begin{lstlisting}
<controllers>
  <air_resistance_controller id="airbot"
      library="build/lib/controllers/air_resistance/libair_resistance">
    <actuators>
      <differential_steering implementation="default"/>
      <range_and_bearing implementation="default"/>
    </actuators>
    <sensors>
      <positioning implementation="default"/>
      <range_and_bearing implementation="medium" medium="rab" show_rays="true"/>
    </sensors>
    <params velocity="15.0"/>
  </air_resistance_controller>
</controllers>
\end{lstlisting}

\subsubsection*{RAB payload contract (important)}
\begin{itemize}
  \item The base class reserves \textbf{RAB byte 0} for a robot radius value (used by wake computations).
  \item For formation/messaging, store your data in \textbf{bytes 1..}.
  \item If your controller writes byte 1, set \texttt{rab\_data\_size} $\ge 2$ for every robot.
\end{itemize}

\section{Which Controller Should I Use?}

\subsection{The controller you should start from}
\textbf{\texttt{air\_resistance\_controller}} is the intended foundation.
It implements the wind+wake impulse pipeline and provides extension hooks for users to build their own behaviors.

\subsection{The other controllers are examples (inherit and extend)}
\texttt{wind\_aware\_air\_resistance\_controller} and \texttt{formation\_template\_controller} are \textbf{example derived controllers}.
They demonstrate how to inherit from \texttt{CAirResistance} and extend it.

\begin{itemize}
  \item \textbf{Wind-aware example}: demonstrates \emph{crosswind compensation} (``crabbing'') to reduce sideways drift.
  \item \textbf{Formation template}: demonstrates \emph{multi-robot coordination} using RAB bytes 1.. while respecting the base wake contract (byte 0 reserved).
\end{itemize}

\section{How to Build a New Controller}
\label{sec:new-controller}

This section describes the minimal steps for adding your own controller on top of \texttt{CAirResistance}.

\subsection{Minimal ControlStep structure (recommended)}
A minimal control-step structure looks like:

\begin{lstlisting}
// 1) Apply wind + reset accumulators + broadcast radius
HandleAerodynamicsPreStep();

// 2) Your logic here: read sensors / RAB, decide cmd speed etc.

// 3) Convert desired motion to impulse
DriveImpulse(cmd_speed_cm_s);

// 4) Apply after physics tick (post-step)
HandleAerodynamicsPostStep();
\end{lstlisting}

\subsection{Important note: DriveImpulse vs SetLinearVelocity}
\textbf{If you want the robot to be driven by the air-resistance model, you should drive using \texttt{DriveImpulse()} and not by directly setting wheel linear velocity.}
The base controller applies wind and drive as impulses in the physics engine (post-step). If you use \texttt{SetLinearVelocity()} as your primary motion command, you are bypassing the intended impulse-based pipeline.

\emph{When is \texttt{SetLinearVelocity()} still okay?} In derived controllers you may use wheel velocities \textbf{only for steering/yaw control}, while still using \texttt{DriveImpulse()} for forward propulsion.

\subsection{Step 1: create a new controller folder}
Create:
\begin{lstlisting}
examples/controllers/my_controller/
\end{lstlisting}

Copy one of the example controllers as a starting point (recommended: \texttt{formation\_template.cpp} because it already shows safe RAB usage).

\subsection{Step 2: add a CMake file}
Add \texttt{examples/controllers/my\_controller/CMakeLists.txt} (pattern matches existing example controllers).
Then register your controller using \texttt{REGISTER\_CONTROLLER(...)}.

\subsection{Step 3: include your controller in the examples build}
Edit \texttt{examples/controllers/CMakeLists.txt} and add:
\begin{lstlisting}
add_subdirectory(my_controller)
\end{lstlisting}

Rebuild:
\begin{lstlisting}
cmake --build build -j
\end{lstlisting}

\subsection{Step 4: reference it from an experiment file}
Create \texttt{examples/experiments/my\_experiment.argos} and load your library:
\begin{lstlisting}
<my_controller id="c1" library="build/lib/controllers/my_controller/libmy_controller">
  ...
  <params velocity="12.0"/>
</my_controller>
\end{lstlisting}

\section{Examples (What They Teach You)}
\label{sec:examples}

Each experiment is designed to teach a small set of configuration concepts:
\textbf{(i)} wind vs drive speed,
\textbf{(ii)} wake alignment vs lateral offset,
\textbf{(iii)} wake width/length tuning,
\textbf{(iv)} how derived controllers extend the base controller.

\subsection{Blocked vs Unblocked (3 e-puck2)}
\textbf{Teaches:} geometric alignment determines shielding.

\textbf{Salient experiment-file ideas}
\begin{itemize}
  \item Wind magnitude is near controller velocity to make differences obvious.
  \item One follower is directly downwind (max overlap); one is laterally offset (reduced overlap).
\end{itemize}

\textbf{What happens if you change wake width/length?}
\begin{itemize}
  \item Increase \texttt{lateral\_reach\_radii}: the lateral robot can become partially blocked too.
  \item Increase \texttt{shadow\_length\_radii}: robots farther behind can still experience shielding.
  \item Increase both and watch as the previously stationary unblocked robot now moves.
\end{itemize}

\noindent\textbf{Run}
\begin{lstlisting}
argos3 -c examples/experiments/airResistance_blocked_vs_unblocked.argos
\end{lstlisting}

\begin{figure}[H]
  \centering
  \includegraphics[width=.85\linewidth]{blocked_vs_unblocked.png}
  \caption{Blocked vs Unblocked: the robot in the wake progresses upwind; the lateral robot experiences less shielding.}
\end{figure}
\FloatBarrier

\subsection{Two Robots --- No Blocking (parallel columns)}
\textbf{Teaches:} lateral separation prevents wake overlap.

\textbf{Salient experiment-file ideas}
\begin{itemize}
  \item Robots are separated in $Y$ enough that wake overlap is minimal.
\end{itemize}

\noindent\textbf{Run}
\begin{lstlisting}
argos3 -c examples/experiments/airResistance_two_no_block.argos
\end{lstlisting}

\begin{figure}[H]

\end{figure}
\FloatBarrier

\subsection{Two Followers Behind a Blocker}
\textbf{Teaches:} partial wake coverage for small lateral offsets.

\textbf{Salient experiment-file ideas}
\begin{itemize}
  \item One blocker upwind; two followers behind with small $Y$ offsets.
\end{itemize}

\textbf{Tweak suggestions}
\begin{itemize}
  \item Increase \texttt{upwind\_gate\_radii} to make the followers stop when they are further away from the leader.
  \item Decrease \texttt{upwind\_gate\_radii} to make the followers stop when they are closer to the leader.
\end{itemize}

\noindent\textbf{Run}
\begin{lstlisting}

\end{lstlisting}

\begin{figure}[H]

  \caption{Two followers behind a single blocker; both can benefit if within the wake width.}
\end{figure}
\FloatBarrier

\subsection{Three in a Row (wake chaining)}
\textbf{Teaches:} wake fades with downwind distance (diminishing benefit).

\textbf{Salient experiment-file ideas}
\begin{itemize}
  \item Robots are placed on the wind axis with increasing downwind distance.
  \item The second robot gets faster as he approaches the stationary leader, the third gets slower as he looses the shielding of the second robot.
\end{itemize}

\textbf{Tweak suggestions}
\begin{itemize}
  \item Increase \texttt{shadow\_length\_radii} to extend the wake further (third robot benefits more).
  \item Decrease \texttt{shadow\_length\_radii} to shorten the wake (third robot may lose shielding).
\end{itemize}

\noindent\textbf{Run}
\begin{lstlisting}
argos3 -c examples/experiments/airResistance_three_in_row.argos
\end{lstlisting}

\section{Tuning Tips}
\begin{itemize}
  \item Set \texttt{magnitude} $\approx$ \texttt{velocity} to make wake differences visually obvious.
  \item To make wake longer/shorter: tune \texttt{shadow\_length\_radii}.
  \item To make wake wider/narrower: tune \texttt{lateral\_reach\_radii}.
  \item If everything drifts downwind: increase \texttt{velocity} or reduce \texttt{magnitude}.
\end{itemize}

% ============================================================
\section{Troubleshooting}
\begin{itemize}
  \item \textbf{Plugins not found:} run from repo root so relative library paths resolve.
  \item \textbf{No wind arrow:} ensure Qt-OpenGL is enabled and the wind user-functions library is loaded in the \texttt{.argos}.
  \item \textbf{Robots not moving:} if wind $\approx$ drive and robots are unshielded, being stationary can be expected.
  \item \textbf{Formation/messaging bugs:} ensure \texttt{rab\_data\_size} is large enough if you use bytes 1.. (byte 0 is reserved).
\end{itemize}
